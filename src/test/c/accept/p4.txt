PATTERN myPattern(Color col,Pattern pattern){
    pattern;
    TURN CH CH CH;
    col pattern;
};

Color red = #FF0000;
Stich myStich = CH;
PATTERN myPattern2(Color col,Stitch st,Stitch st2){
    st st2 st;
    TURN CH CH CH;
    col CH CH CH;
};

BEGIN CROCHET
    myPattern(red,CH);
    TURN CH;
    myPattern(red,DC);
    REPEAT(myPattern2(red,CD,myStich), 3) CH;
    MIRROR(myPattern2(red,CD,myStich), 3) CH;
END CROCHET
